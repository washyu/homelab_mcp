    ```python
    def _resolve_ssh_credentials_with_binding(
        hostname: str,
        username: str | None = None,
        password: str | None = None,
        key_path: str | None = None,
        port: int = 22,
        *,
        credential_id: str | None = None,
    ) -> tuple[SSHCredentials, str | None]:
        """Phase 38.1 R3 helper — resolves SSH credentials AND returns the registry
        entry's credential_id (when keyring-resolved) or None (when Tier-1 explicit-args).

        Discovery's auto-bind side effect calls this instead of `resolve_ssh_credentials`
        so the upsert path can record `ssh_credential_id` on the new sitemap row.

        Behaviour:
            - When ``credential_id`` is supplied: Tier-0 short-circuit; returns
              that UUID alongside the credentials.
            - When explicit ``password`` or ``key_path`` is supplied: Tier-1
              path; returns ``(creds, None)`` because no registry entry was used.
            - When no credentials supplied (canonical workflow): Tier-2 path
              walks the registry; returns the matched entry's ``credential_id``.

        This wrapper does the registry scan AGAIN to capture the entry's UUID
        because ``resolve_ssh_credentials`` doesn't expose it. Cost is a single
        JSON file read — acceptable on the discovery path.
        """
        # Tier-0: caller-supplied UUID short-circuit. resolve_ssh_credentials
        # handles all D-11/D-12/D-13 logic; return the same UUID back.
        if credential_id is not None:
            creds = resolve_ssh_credentials(
                hostname, username, password, key_path, port,
                credential_id=credential_id,
            )
            return creds, credential_id

        # Tier-1: explicit-args path — no registry hit, no binding to record.
        if password is not None or key_path is not None:
            creds = resolve_ssh_credentials(hostname, username, password, key_path, port)
            return creds, None

        # Tier-2: keyring registry path. Scan registry for matching SSH entry
        # BEFORE calling resolve_ssh_credentials so we know the credential_id
        # of the entry we're about to use. Match logic mirrors resolve_ssh_credentials's
        # internal scan (which currently lives at lines ~124-160 in the post-Plan-05 file).
        from .credential_store import list_credentials  # noqa: PLC0415

        entries = list_credentials(credential_type="ssh")
        # Filter: hostname-exact, then optional username narrowing.
        matching = [e for e in entries if e.get("hostname") == hostname]
        if username is not None:
            matching = [e for e in matching if e.get("username") == username]
        # If exactly one matching entry, capture its credential_id; otherwise leave None
        # (resolve_ssh_credentials will raise/handle ambiguity via existing logic).
        used_id: str | None = None
        if len(matching) == 1:
            used_id = matching[0].get("credential_id")  # legacy entries → None

        creds = resolve_ssh_credentials(hostname, username, password, key_path, port)
        return creds, used_id
    ```
    The Tier-2 scan duplicates a single registry-read (a JSON file load that's already cached at the OS page level — milliseconds at worst). Cleaner than restructuring `resolve_ssh_credentials` to return a tuple (which would break every existing caller).
