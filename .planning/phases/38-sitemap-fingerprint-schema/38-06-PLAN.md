```python
@pytest.mark.integration
@pytest.mark.asyncio
async def test_discover_populates_fingerprint_against_docker_phase38(
    self, test_container, tmp_path
):
    """Phase 38 SC-1 + SC-2: live SSH discovery against Docker container populates fingerprint sub-dict.

    Proves the end-to-end chain: ssh_discover_system probes (Plan 01) →
    parse_discovery_output (Plan 02) → SQLiteAdapter store/get round-trip (Plan 03).
    Reuses the existing Ubuntu test container at localhost:2222.
    """
    from homelab_mcp.sitemap import NetworkSiteMap, discover_and_store

    db_path = str(tmp_path / "phase38_fingerprint.db")
    sitemap = NetworkSiteMap(db_path=db_path, db_type="sqlite")

    result_str = await discover_and_store(
        sitemap,
        hostname=test_container["hostname"],
        username=test_container["admin_user"],
        password=test_container["admin_pass"],
        port=test_container["port"],
    )
    # Parse the discovery result first to confirm it succeeded
    import json as _json
    result = _json.loads(result_str) if isinstance(result_str, str) else result_str
    assert result.get("status") in ("success", "ok"), f"Discovery should succeed; got: {result}"

    devices = sitemap.get_all_devices()
    assert len(devices) >= 1, f"Expected at least one device row; got {len(devices)}"
    # Find the row by hostname (the container's reported hostname, not necessarily 'localhost')
    target = devices[0]  # only one device expected from this test
    fp = target.get("fingerprint")
    assert fp is not None, f"fingerprint should be populated; row keys: {list(target.keys())}"
    assert isinstance(fp, dict), f"fingerprint should be a dict (decoded from JSON), got {type(fp)}"

    # SC-1: kernel_name from `uname -s`
    assert fp.get("kernel_name") == "Linux", f"Expected kernel_name=Linux, got {fp.get('kernel_name')!r}"

    # SC-1: kernel_version from `uname -r` — non-empty string matching typical kernel-version shape
    kv = fp.get("kernel_version")
    assert kv, f"kernel_version should be populated, got {kv!r}"
    assert isinstance(kv, str) and len(kv) >= 3, f"kernel_version unexpectedly short: {kv!r}"

    # SC-1: os_name + os_version from /etc/os-release PRETTY_NAME / VERSION_ID
    # The container is Ubuntu-based; PRETTY_NAME typically contains "Ubuntu" or the underlying base image name.
    # Be lenient: assert os_name is a non-empty string.
    os_name = fp.get("os_name")
    assert os_name and isinstance(os_name, str), f"os_name should be populated, got {os_name!r}"

    # SC-1: package_fingerprint from `LC_ALL=C dpkg -l | sort | sha256sum`
    pkg_fp = fp.get("package_fingerprint")
    assert pkg_fp, f"package_fingerprint should be populated on Debian/Ubuntu, got {pkg_fp!r}"
    assert pkg_fp.startswith("sha256:"), f"package_fingerprint should start with 'sha256:', got {pkg_fp!r}"
    # Verify the digest is exactly 64 hex chars after the prefix
    digest = pkg_fp[len("sha256:"):]
    assert len(digest) == 64, f"Expected 64-char sha256 hex digest, got {len(digest)} chars: {digest!r}"
    assert all(c in "0123456789abcdef" for c in digest.lower()), f"Digest should be hex: {digest!r}"
```
